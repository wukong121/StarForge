cmake_minimum_required(VERSION 3.20)
project(StarForgeBWAPI LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The game bridge requires Windows and MSVC; Python works on other platforms.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use -A Win32. StarCraft 1.16.1 loads 32-bit x86 DLLs, including on ARM Windows.")
endif()
set(BWAPI_ROOT "" CACHE PATH "BWAPI 4.4.0 SDK directory containing include/ and lib/")
find_path(BWAPI_INCLUDE_DIR BWAPI.h PATHS "${BWAPI_ROOT}/include" NO_DEFAULT_PATH)
find_library(BWAPI_LIBRARY NAMES BWAPI PATHS "${BWAPI_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT BWAPI_INCLUDE_DIR OR NOT BWAPI_LIBRARY)
  message(FATAL_ERROR "BWAPI 4.4.0 SDK not found. Set -DBWAPI_ROOT=<SDK>; expected include/BWAPI.h and lib/BWAPI.lib.")
endif()
add_library(StarForge SHARED StarForge.cpp)
target_compile_features(StarForge PRIVATE cxx_std_17)
target_compile_definitions(StarForge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(StarForge PRIVATE /W4 /EHsc)
target_include_directories(StarForge PRIVATE "${BWAPI_INCLUDE_DIR}")
target_link_libraries(StarForge PRIVATE "${BWAPI_LIBRARY}" ws2_32)
# Official release SDK uses the release MSVC runtime. Build Release, not Debug.
set_property(TARGET StarForge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
