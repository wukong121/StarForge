cmake_minimum_required(VERSION 3.20)
project(StarForgeBWAPI LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The game bridge requires Windows and MSVC; Python works on other platforms.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use -A Win32. StarCraft 1.16.1 loads 32-bit x86 DLLs, including on ARM Windows.")
endif()
set(BWAPI_ROOT "" CACHE PATH "BWAPI 4.4.0 SDK directory containing include/")
set(BWAPI_LIBRARY "" CACHE FILEPATH "Release Win32 BWAPILIB.lib built from the matching SDK")
find_path(BWAPI_INCLUDE_DIR BWAPI.h PATHS "${BWAPI_ROOT}/include" NO_DEFAULT_PATH)
if(NOT BWAPI_INCLUDE_DIR OR NOT EXISTS "${BWAPI_LIBRARY}")
  message(FATAL_ERROR "Set BWAPI_ROOT to the 4.4.0 SDK and BWAPI_LIBRARY to a matching Release Win32 BWAPILIB.lib. The official SDK ships library source, not a prebuilt .lib. See docs/native-build.md.")
endif()
add_library(StarForge SHARED StarForge.cpp)
add_executable(StarForgeProbe BridgeProbe.cpp)
foreach(target IN ITEMS StarForge StarForgeProbe)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(${target} PRIVATE /W4 /EHsc)
  target_include_directories(${target} PRIVATE "${BWAPI_INCLUDE_DIR}")
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endforeach()
target_link_libraries(StarForge PRIVATE "${BWAPI_LIBRARY}" ws2_32)
